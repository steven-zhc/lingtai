/**
 * Acceptance for S5: a sandboxed Codex run goes through srt with Codex's own sandbox off, the operator's Codex
 * profile denies on the run, and the Codex home's configuration write-denied (ADR 0127 §1, §15, §16).
 *
 * Unit: nothing here starts srt or Codex. `node:fs` is replaced for the fixture paths only (the hook wiring, the
 * operator's `config.toml`, a linked worktree's `.git`), the way `interruption.test.ts` does it, `spawn` is a
 * stand-in that records its call, and `request.sandbox.wrap` is a fake `Sandbox.wrap` that records its input.
 * The recorded wrap input is then put through `@lingtai/sandbox`'s own `resolvedPolicy`, so the claims about what
 * srt is handed are made against the policy the package would really build, not against our reading of it.
 *
 * The fixture is the real layout: the worktree's `.git` is a file pointing into `<stateDir>/repos/p.git`, so
 * today's argv carries `--add-dir` and its absence in the sandboxed argv means something. The project cache is
 * `cacheDir(host, 'p')` = `/state/cache/p`, beside the mirror `mirrorDir(host, 'p')` = `/state/repos/p.git`.
 */

const files = new Map<string, string>()
const present = new Set<string>()

const enoent = (path: string) => Object.assign(new Error(`ENOENT: no such file or directory, open '${path}'`), { code: 'ENOENT' })

vi.mock('node:fs', async (importOriginal) => {
  const real = await importOriginal<typeof import('node:fs')>()
  const owned = (p: unknown) => typeof p === 'string' && (p.startsWith('/home/op') || p.startsWith('/work/') || p.startsWith('/state') || p.startsWith('/opt/codex-home'))
  return {
    ...real,
    readFileSync: ((path: unknown, ...rest: unknown[]) => {
      if (typeof path === 'string' && files.has(path)) return files.get(path)
      if (owned(path)) throw enoent(String(path))
      return (real.readFileSync as (...a: unknown[]) => unknown)(path, ...rest)
    }) as typeof real.readFileSync,
    statSync: ((path: unknown, ...rest: unknown[]) => {
      if (path === `${CWD}/.git`) return { isDirectory: () => false, isFile: () => true }
      if (owned(path)) throw enoent(String(path))
      return (real.statSync as (...a: unknown[]) => unknown)(path, ...rest)
    }) as typeof real.statSync,
    existsSync: ((path: unknown) =>
      typeof path === 'string' && (present.has(path) || files.has(path))
        ? true
        : owned(path)
          ? false
          : real.existsSync(path as string)) as typeof real.existsSync,
    mkdirSync: ((path: unknown, ...rest: unknown[]) =>
      owned(path) ? undefined : (real.mkdirSync as (...a: unknown[]) => unknown)(path, ...rest)) as typeof real.mkdirSync,
  }
})

const wiring = () => renderSettings({ runId: 'r1', hookBinary: '/opt/lingtai/lingtai-hook' })

/** A profile naming `default_permissions = "mine"` whose filesystem table denies each key. */
const profileDenying = (...keys: string[]) =>
  `default_permissions = "mine"\n[permissions.mine]\nextends = ":workspace"\n[permissions.mine.filesystem]\n` +
  keys.map((k) => `${JSON.stringify(k)} = "deny"\n`).join('')

const setConfig = (text: string | null, home = CODEX_HOME) => {
  const path = `${home}/config.toml`
  if (text === null) files.delete(path)
  else files.set(path, text)
}

// ── the stand-ins ───────────────────────────────────────────────────────────────────────────────────────────

type SpawnCall = { command: string; args: readonly string[]; options: Record<string, unknown> }

const child = (closes: boolean) => {
  const c = Object.assign(new EventEmitter(), {
    pid: undefined,
    stdout: new PassThrough(),
    stderr: new PassThrough(),
    kill: vi.fn(),
  })
  if (closes) setImmediate(() => c.emit('close', 0))
  return c
}

const SRT = '/usr/local/bin/srt'
const POLICY_VARS = { XDG_CACHE_HOME: `${CACHE}/xdg-cache`, DO_NOT_TRACK: '1' }

const harness = (
  opts: {
    decision?: RunSandbox['decision'] | 'absent'
    env?: Record<string, string>
    wrap?: RunSandbox['wrap']
    closes?: boolean
  } = {},
) => {
  const wrapped: WrapInput[] = []
  const spawned: SpawnCall[] = []
  const stops: string[] = []
  const decision = opts.decision ?? { kind: 'sandboxed', project_additions: { network: ['example.com'] } }
  const sandbox: RunSandbox | undefined =
    decision === 'absent'
      ? undefined
      : {
          decision,
          project: 'p',
          mirror: MIRROR,
          worktree: CWD,
          runDir: `${STATE}/runs/r1/claude`,
          operatorClaudeDir: `${HOME}/.claude`,
          hostHome: HOME,
          read: [],
          wrap:
            opts.wrap ??
            ((input) =>
              Effect.sync(() => {
                wrapped.push(input)
                return { argv: [SRT, '--settings', '/tmp/srt-x.json', '--', ...input.command], env: { ...input.env, ...POLICY_VARS }, commandId: 'cmd-1' }
              })),
        }
  const request: RunRequest = {
    runId: 'r1',
    cwd: CWD,
    prompt: 'do it',
    settingsPath: SETTINGS,
    env: opts.env ?? { HOME, PATH: '/usr/bin', KEEP: '1' },
    limits: { wallMs: 60_000 },
    ...(sandbox === undefined ? {} : { sandbox }),
  }
  const options = {
    binary: 'codex',
    spawn: ((command: string, args: readonly string[], options: Record<string, unknown>) => {
      spawned.push({ command, args, options })
      return child(opts.closes ?? true)
    }) as unknown as typeof spawn,
    stopProcesses: (marker: string) =>
      Effect.sync(() => {
        stops.push(marker)
      }),
  }
  return { request, options, wrapped, spawned, stops }
}

const run = (h: ReturnType<typeof harness>) => Effect.runPromise(codexRun(h.options, h.request))

/** The host the recorded wrap input is resolved against: the same `HOME` and `stateDir` the fixture uses. */
const HOST: HostFacts = { platform: 'linux', home: HOME, tmpdir: '/tmp', stateDir: STATE, path: ['/usr/bin'] }

const policyOf = (input: WrapInput) =>
  resolvedPolicy(HOST, {
    project: input.project,
    worktree: input.worktree,
    mirror: input.mirror,
    project_additions: input.project_additions,
    caller_additions: input.caller_additions,
  })

/** Whether `path` is one of `entries` or lies inside one. */
const coveredBy = (path: string, entries: readonly string[]) =>
  entries.some((e) => path === e || path.startsWith(e.endsWith('/') ? e : `${e}/`))

const flagValue = (args: readonly string[], flag: string) => args[args.indexOf(flag) + 1]

// ── Want: sandboxed → through Sandbox.wrap, with --sandbox danger-full-access ────────────────────────────────

describe('Want: a sandboxed Codex run goes through Sandbox.wrap with Codex’s own sandbox off', () => {
  it('Done when 3: one wrap whose command is codex exec … --sandbox danger-full-access, no profile -c, no --add-dir', async () => {
    setConfig(profileDenying('~/secret'))
    const h = harness()
    await run(h)

    expect(h.wrapped).toHaveLength(1)
    const input = h.wrapped[0]!
    const cmd = input.command
    expect(cmd[0]).toBe('codex')
    expect(cmd[1]).toBe('exec')
    expect(flagValue(cmd, '--sandbox')).toBe('danger-full-access')
    expect(cmd.filter((a) => a === '--sandbox')).toHaveLength(1)
    expect(cmd.some((a) => a.startsWith('default_permissions='))).toBe(false)
    expect(cmd.some((a) => a.startsWith('permissions.'))).toBe(false)
    expect(cmd.some((a) => a.startsWith('profiles'))).toBe(false)
    expect(cmd).not.toContain('--add-dir')
    // Everything else the run needs is still there.
    expect(flagValue(cmd, '--cd')).toBe(CWD)
    expect(cmd).toContain('approval_policy=never')
    expect(cmd).not.toContain('--dangerously-bypass-approvals-and-sandbox')
    expect(cmd.at(-2)).toBe('--')
    expect(cmd.at(-1)).toBe('do it')

    // The run's own inputs, not the conductor's guesses.
    expect(input.project).toBe('p')
    expect(input.worktree).toBe(CWD)
    expect(input.mirror).toBe(MIRROR)
    expect(input.project_additions).toEqual({ network: ['example.com'] })
  })

  it('Done when 3 / Watch out: the spawn is the wrap’s argv and env, with no shell; srt’s env is the only env', async () => {
    const h = harness()
    await run(h)

    expect(h.spawned).toHaveLength(1)
    const s = h.spawned[0]!
    expect(s.command).toBe(SRT)
    expect(s.args.slice(0, 4)).toEqual(['--settings', '/tmp/srt-x.json', '--', 'codex'])
    expect(s.options.shell ?? false).toBe(false)
    expect(s.options.cwd).toBe(CWD)
    // Exactly what `wrap` returned: the caller's filtered env plus the policy's variables, marker included.
    const wrappedEnv = { ...h.wrapped[0]!.env, ...POLICY_VARS }
    expect(s.options.env).toEqual(wrappedEnv)
  })

  it('Done when 3 / 6: the env handed to wrap carries LINGTAI_CODEX_PROCESS_MARKER and the run’s filtered env', async () => {
    const h = harness()
    await run(h)
    const env = h.wrapped[0]!.env

    expect(env.LINGTAI_CODEX_PROCESS_MARKER).toMatch(/\S{8,}/)
  })
})

// ── Done when 8: the exported conversion ─────────────────────────────────────────────────────────────────────

/**
 * The ticket asks for *an exported function*; this names it `absoluteProfileDenies(verdict, { home, cwd })` in
 * `codex.ts`. Its answer may be a list, `{ paths }`, or a refusal carrying `why` (as a value or a thrown error);
 * `normalise` reads any of those, so only the name and the arguments are pinned.
 */
type Converted = { ok: true; paths: string[] } | { ok: false; why: string }
const normalise = (call: () => unknown): Converted => {
  let out: unknown
  try {
    out = call()
    if (Effect.isEffect(out)) {
      const exit = Effect.runSyncExit(out as Effect.Effect<unknown, unknown>)
      if (Exit.isFailure(exit)) return { ok: false, why: JSON.stringify(exit.cause) + String(exit.cause) }
      out = exit.value
    }
  } catch (e) {
    return { ok: false, why: e instanceof Error ? e.message : String(e) }
  }
  const o = out as Record<string, unknown> | unknown[]
  if (Array.isArray(o)) return { ok: true, paths: o as string[] }
  if (o && typeof o === 'object') {
    if (Array.isArray(o.paths)) return { ok: true, paths: o.paths as string[] }
    for (const k of ['why', 'reason', 'sentence', 'message', 'detail']) {
      if (typeof o[k] === 'string') return { ok: false, why: o[k] as string }
    }
  }
  return { ok: false, why: `unrecognised answer ${JSON.stringify(out)}` }
}

describe('Done when 8: an exported function turns a UserPermissions verdict into absolute deny paths', () => {
  const convert = (verdict: codex.UserPermissions, at = { home: HOME, cwd: CWD }) => {
    const fn = (codex as unknown as Record<string, unknown>).absoluteProfileDenies
    expect(typeof fn, 'codex.ts exports absoluteProfileDenies').toBe('function')
    return normalise(() => (fn as (v: unknown, a: unknown) => unknown)(verdict, at))
  }

  it('resolves ~/ against home, a relative key against cwd, keeps absolute and globbed keys as written', () => {
    const verdict = userPermissionsFrom(profileDenying('~/secret', '/abs/x', 'rel/y', '../up', '/vault/**/*.pem'))
    expect(convert(verdict)).toEqual({
      ok: true,
      paths: [`${HOME}/secret`, '/abs/x', `${CWD}/rel/y`, '/work/up', '/vault/**/*.pem'],
    })
  })

  it('answers no paths for none and for kept', () => {
    expect(convert({ kind: 'none' })).toEqual({ ok: true, paths: [] })
    expect(convert({ kind: 'kept', why: 'sandbox_mode' })).toEqual({ ok: true, paths: [] })
  })

  it('refuses a :special key by name', () => {
    const answer = convert({ kind: 'denies', paths: ['/abs/x', ':tmpdir'] })
    expect(answer.ok).toBe(false)
    expect(answer.ok === false && answer.why).toContain(':tmpdir')
  })
})

        { denies: ['/x'], overrideIn: 'dev.local', profiles: { 'dev.local': { model: 'm' } } },
        { outputSchemaPath: '/tmp/schema.json' },
      ]) {
        unsandboxedArgvs.push(codexArgv(base, 'p', { settings: wiring(), sandbox, ...extra }))
        unsandboxedArgvs.push(codexArgv({ ...base, model: 'gpt-5.6' }, 'p', { settings: null, sandbox, ...extra }))
      }
      const runtime = createCodexRuntime({ sandbox })
      unsandboxedArgvs.push([...runtime.invocation!(base).args])
      const refused = harness({ decision: { kind: 'refused', why: 'no srt' } }).request.sandbox!
      unsandboxedArgvs.push([...runtime.invocation!({ ...base, sandbox: refused }).args])
    }
    for (const args of unsandboxedArgvs) expect(args.join(' ')).not.toContain('danger-full-access')

    // The positive control: the branch that may carry it does.
    const sandboxed = harness().request.sandbox!
    const shown = createCodexRuntime().invocation!({ ...base, sandbox: sandboxed })
    expect(flagValue(shown.args, '--sandbox')).toBe('danger-full-access')
  })
})
