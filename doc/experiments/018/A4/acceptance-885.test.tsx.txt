 *

 * ticket names:

const ITEM = 'ticket-lingtai-885'
const at = (path: string) => `/task/${encodeURIComponent(ITEM)}${path}`
const runId = (i: number) => `pass-${String(i).padStart(8, '0')}-0000-0000-0000-000000000000`
const base = (i: number) => `${i}${i}${i}${i}bad`.padEnd(40, '0')
const head = (i: number) => `${i}${i}${i}${i}fed`.padEnd(40, '0')
const turns = (i: number) => 10 * i + 7
const WORKTREE = (i: number) => `/Users/someone/.lingtai/worktrees/lingtai/${runId(i)}`
const WAIT_RAW = 'WAIT-RAW: the reviewer refused, and this is the whole of what it said'

let seq = 0n
function e(streamId: string, type: string, data: unknown, when = '2026-11-04T04:00:00Z'): Envelope {
  seq += 1n
  return {
    seq,
    streamId,
    version: 1,
    type,
    schemaVer: 1,
    data,
    actor: 'conductor',
    causation: null,
    at: new Date(when),
  } as Envelope
}

function verdict(name: string, evidence: string, v: 'accepted' | 'refused', onSha: string) {
  return {
    plugin: 'run',
    name,
    onSha,
    evidence,
    yields: 'verdict',
    result: { _tag: 'Completed', assessment: { verdict: v, findings: [] } },
  }
}

/** One attempt's own stream: started, prompted, a file, proposed, one review, finished. */
function attemptStream(i: number, { refused = false, finished = true } = {}): Envelope[] {
  const r = runId(i)
  return [
    e(r, 'RunStarted', { baseSha: base(i) }),
    e(r, 'RunPrompted', { prompt: `PROMPT-OF-ATTEMPT-${i}\n`, promptVersion: `ticket@19${i}`, bytes: 4100 + i }),
    e(r, 'RunTouchedFile', { path: `${WORKTREE(i)}/apps/board/src/file-${i}.ts`, op: 'edit' }),
    e(r, 'RunProposedCompletion', { headSha: head(i) }),
    e(r, 'StepStarted', { step: 'proposed', round: 0 }),
    e(r, 'StepEnded', {
      outcome: {
        step: 'proposed',
        round: 0,
        actions: [verdict('review', `VERDICT-EVIDENCE-${i}`, refused ? 'refused' : 'accepted', head(i))],
      },
    }),
    ...(finished ? [e(r, 'RunFinished', { turns: turns(i), durationMs: 60_000 * i, exitCode: 0 })] : []),
  ]
}

interface Scenario {
  /** How many attempts. */
  n: number
  /** The attempt whose review refused — the one `standing.deciding` will name. */
  refused?: number
  /** Whether the ticket is stopped at a wait (on the last attempt) now. */
  waiting?: boolean
}

/** The ticket's own stream and every claim's, keyed by stream id. */
function log({ n, refused, waiting = false }: Scenario): Map<string, Envelope[]> {
  const own: Envelope[] = []
  const streams = new Map<string, Envelope[]>()
  for (let i = 1; i <= n; i += 1) {
    own.push(e(ITEM, 'TicketClaimed', { runId: runId(i), worker: 'conductor:1', title: null, kind: null }))
    streams.set(runId(i), attemptStream(i, { refused: i === refused }))
    if (i < n || !waiting) own.push(e(ITEM, 'TicketReleased', { runId: runId(i), reason: 'released', cause: null }))
  }
  if (waiting && n > 0) {
    own.push(
      e(ITEM, 'WaitStarted', {
        waitId: 'wait-885',
        reason: {
          _tag: 'CheckRefused',
          run: { runId: runId(n) },
          step: 'proposed',
          action: 'review',
          evidence: WAIT_RAW,
          findings: [],
        },
      }),
    )
  }
  streams.set(ITEM, own)
  return streams
}

/** What `loadTask` would have reduced, built from the same log with today's exports. */
function detail(s: Scenario): TaskDetail {
  const streams = log(s)
  const own = streams.get(ITEM)!
  const claims = reduceTicket(own).claims
  const runs: AttemptView[] = claims.map((c, i) => presentAttemptEvents(c, i + 1, streams.get(c.runId) ?? []))
  return {
    taskId: ITEM,
    waiting: { previous: null, next: null },
    unpushedMirror: null,
    standing: standingOf(own, runs),
    want: null,
    ticket: {
      project: 'lingtai',
      ref: '885',
      title: 'TICKET-TITLE the record is one tab per subject',
      kind: 'feature',
      labels: ['feature'],
      url: 'https://github.com/example/lingtai/issues/885',
      compareUrl: null,
      body: 'TICKET-BODY',
      found: true,
      askedGitHub: true,
      problem: null,
    },
    runs,
    discussions: [],
    outgoing: null,
    queued: null,
    held: null,
    totals: totalsOf(runs),
    spendCurrency: { lag: 0 },
    history: [],
  } as unknown as TaskDetail
}

function serve(s: Scenario) {

  vi.mocked(eventStore.read).mockImplementation((async (stream: string) => streams.get(stream) ?? []) as never)
}

type Page = (props: { params: Promise<Record<string, string>> }) => Promise<unknown>

async function renderChild(page: Page, params: Record<string, string>): Promise<string> {
  const element = (await page({ params: Promise.resolve({ id: ITEM, ...params }) })) as ReactNode
  return renderToStaticMarkup(<>{element}</>)
}

async function attemptTab(n: number | string): Promise<string> {
  nav.segments = ['attempt', String(n)]
  return renderChild((await attemptRoute()).default as Page, { n: String(n) })
}
async function ticketTab(): Promise<string> {
  nav.segments = ['ticket']
  return renderChild((await ticketRoute()).default as Page, {})
}
async function defaultTab(): Promise<string> {
  nav.segments = []
  return renderChild((await defaultRoute()).default as Page, {})
}

/** The layout's body around a child's markup, at the path `segments` names. */
async function whole(task: TaskDetail, segments: string[], child = '<p>CHILD</p>'): Promise<string> {
  nav.segments = segments
  const { TaskBody } = await layoutRoute()
  const Body = TaskBody as (props: { task: TaskDetail; children?: ReactNode }) => ReactElement
  return renderToStaticMarkup(
    <Body task={task}>
      <div dangerouslySetInnerHTML={{ __html: child }} />
    </Body>,
  )
}

// ---------------------------------------------------------------- an HTML tree, without a DOM

interface El {
  tag: string
  attrs: Record<string, string>
  kids: (El | string)[]
  parent: El | null
}
const VOID = new Set(['area', 'base', 'br', 'col', 'embed', 'hr', 'img', 'input', 'link', 'meta', 'source', 'track', 'wbr'])
const decode = (s: string) =>
  s
    .replaceAll('&lt;', '<')
    .replaceAll('&gt;', '>')
    .replaceAll('&quot;', '"')
    .replaceAll('&#x27;', "'")
    .replaceAll('&#39;', "'")
    .replaceAll('&nbsp;', ' ')
    .replaceAll('&amp;', '&')

function parse(html: string): El {
  const root: El = { tag: '#root', attrs: {}, kids: [], parent: null }
  let cur = root
  const re = /<!--[\s\S]*?-->|<\/([a-zA-Z0-9-]+)\s*>|<([a-zA-Z][a-zA-Z0-9-]*)((?:\s+[^\s=>/]+(?:="[^"]*")?)*)\s*(\/?)>|([^<]+)/g
  for (const m of html.matchAll(re)) {
    if (m[0].startsWith('<!--')) continue
    if (m[1] !== undefined) {
      let n: El | null = cur
      while (n !== null && n.tag !== m[1].toLowerCase()) n = n.parent
      if (n?.parent) cur = n.parent
    } else if (m[2] !== undefined) {
      const attrs: Record<string, string> = {}
      for (const a of (m[3] ?? '').matchAll(/([^\s=]+)(?:="([^"]*)")?/g)) attrs[a[1]!] = decode(a[2] ?? '')
      const el: El = { tag: m[2].toLowerCase(), attrs, kids: [], parent: cur }
      cur.kids.push(el)
      if (!VOID.has(el.tag) && m[4] !== '/') cur = el
    } else if (m[5] !== undefined) cur.kids.push(decode(m[5]))
  }
  return root
}
const text = (el: El): string => el.kids.map((k) => (typeof k === 'string' ? k : text(k))).join('')
function all(el: El, pred: (e: El) => boolean, out: El[] = []): El[] {
  for (const k of el.kids) {
    if (typeof k === 'string') continue
    if (pred(k)) out.push(k)
    all(k, pred, out)
  }
  return out
}
function closest(el: El, pred: (e: El) => boolean): El | null {
  for (let p = el.parent; p !== null; p = p.parent) if (pred(p)) return p
  return null
}
const squash = (s: string) => s.replace(/\s+/g, ' ').trim()

/** The `<details>` whose `<summary>` reads `name` first, case aside. */
function section(root: El, name: string): El | undefined {
  const summary = all(
    root,
    (e) => e.tag === 'summary' && squash(text(e)).toLowerCase().startsWith(name.toLowerCase()),
  )[0]
  return summary?.parent?.tag === 'details' ? summary.parent : undefined
}

const TAB_HREF = new RegExp(`^/task/${encodeURIComponent(ITEM)}/(attempt/\\d+|ticket)$`)

interface TabRow {
  /** `ticket` or `attempt/N`, for each tab link, in page order. */
  tabs: string[]
  labels: string[]
  selected: string[]
  /** The same keys, for the links inside *earlier attempts*. */
  earlier: string[]
}
function tabRow(html: string): TabRow {
  const root = parse(html)
  const isEarlier = (e: El) => /earlier attempts/i.test(text(e)) && e.tag !== '#root'
  const links = all(
    root,
    (e) => e.tag === 'a' && TAB_HREF.test(e.attrs.href ?? '') && /^(Ticket|Attempt \d+)\b/.test(squash(text(e))),
  )
  const keyOf = (a: El) => a.attrs.href!.replace(`/task/${encodeURIComponent(ITEM)}/`, '')
  // The smallest element that says *earlier attempts* and holds a link.
  const earlierBox = all(root, isEarlier)
    .filter((e) => links.some((a) => closest(a, (p) => p === e)))
    .at(-1)
  const inEarlier = (a: El) => earlierBox !== undefined && closest(a, (p) => p === earlierBox) !== null
  const tabs = links.filter((a) => !inEarlier(a))
  return {
    tabs: tabs.map(keyOf),
    labels: tabs.map((a) => squash(text(a))),
    selected: tabs.filter((a) => a.attrs['aria-current'] === 'page' || a.attrs['aria-selected'] === 'true').map(keyOf),
    earlier: links.filter(inEarlier).map(keyOf),
  }
}

describe('Want: one tab per subject — Ticket, then Attempt N for each attempt, newest first; each its own URL', () => {
  it('Want + Edge "Tab facts": the tab row is links — Ticket first, then Attempt 3, 2, 1 — each with its outcome, and Ticket with its event count', async () => {
    const row = tabRow(await whole(detail({ n: 3 }), ['attempt', '2']))
    expect(row.tabs).toEqual(['ticket', 'attempt/3', 'attempt/2', 'attempt/1'])
    expect(row.labels[0]).toMatch(/^Ticket\b.*\d+ events?/)
    expect(row.labels[1]).toMatch(/^Attempt 3\b.*\bfinished\b/)
    expect(row.earlier).toEqual([])
  })

  it('Want: none of the five old tabs (History, Diff, Findings, Attempts) is in the row', async () => {
    const html = await whole(detail({ n: 2 }), [])
    const root = parse(html)
    expect(all(root, (e) => e.attrs.role === 'tabpanel')).toEqual([])
    const tabWords = all(root, (e) => e.attrs.role === 'tab' || e.tag === 'button').map((b) => squash(text(b)))
    for (const old of ['History', 'Diff', 'Findings', 'Attempts']) {
      expect(tabWords.some((w) => w.toLowerCase().startsWith(old.toLowerCase()))).toBe(false)
    }
    expect(tabRow(html).tabs).toContain('ticket')
  })

  it('Done when 6: with six attempts, the three newest have tabs, the deciding (1) and the open one (2) have tabs though older, and 3 is under *earlier attempts*', async () => {
    const row = tabRow(await whole(detail({ n: 6, refused: 1, waiting: true }), ['attempt', '2']))
    expect(row.tabs[0]).toBe('ticket')
    expect([...row.tabs].sort()).toEqual(['attempt/1', 'attempt/2', 'attempt/4', 'attempt/5', 'attempt/6', 'ticket'])
    expect(row.earlier).toEqual(['attempt/3'])
  })

  it('Done when 6: at most five attempt tabs, and with nothing older open or deciding the rest are all earlier', async () => {
    const row = tabRow(await whole(detail({ n: 6 }), ['attempt', '6']))
    expect(row.tabs).toEqual(['ticket', 'attempt/6', 'attempt/5', 'attempt/4'])
    expect([...row.earlier].sort()).toEqual(['attempt/1', 'attempt/2', 'attempt/3'])
  })

  it('Want "each tab is its own URL" + Edge "The selected tab": the tab the path names is the one marked, and only it', async () => {
    const t = detail({ n: 3 })
    expect(tabRow(await whole(t, ['attempt', '2'])).selected).toEqual(['attempt/2'])
    expect(tabRow(await whole(t, ['ticket'])).selected).toEqual(['ticket'])
  })

  it('Edge "The selected tab": at /task/<id> the default tab is marked — the deciding attempt, not the newest', async () => {
    const row = tabRow(await whole(detail({ n: 3, refused: 2, waiting: true }), []))
    expect(row.selected).toEqual(['attempt/2'])
  })
})

describe('Done when 3: an attempt tab’s order', () => {
  it('summary line, base/head line, Steps (rail, then verdicts), the reasoning (deciding only), then Changes, Prompt, Recipe, Run log, Events — closed', async () => {
    serve({ n: 3, refused: 2, waiting: true })
    const html = await attemptTab(2)
    const root = parse(html)
    const flat = text(root)

    const summary = flat.indexOf(`${turns(2)} turns`)
    const meta = flat.indexOf(base(2).slice(0, 7))
    const steps = flat.indexOf('Steps')
    const rail = html.indexOf('class="segs"')
    const evidence = flat.indexOf('VERDICT-EVIDENCE-2')
    const reasoning = flat.indexOf(WAIT_RAW)
    expect(summary).toBeGreaterThanOrEqual(0)
    expect(meta).toBeGreaterThan(summary)
    expect(steps).toBeGreaterThan(meta)
    expect(rail).toBeGreaterThan(html.indexOf('Steps'))
    expect(evidence).toBeGreaterThan(steps)
    expect(reasoning).toBeGreaterThan(evidence)

    const names = ['Changes', 'Prompt', 'Recipe', 'Run log', 'Events']
    const sections = names.map((n) => section(root, n))
    sections.forEach((s, i) => expect(s, `the ${names[i]} section`).toBeDefined())
    const positions = sections.map((s) => flat.indexOf(text(s!)))
    expect(positions[0]).toBeGreaterThan(reasoning)
    expect([...positions].sort((a, b) => a - b)).toEqual(positions)
    for (const [i, s] of sections.entries()) expect(s!.attrs.open, `${names[i]} is closed`).toBeUndefined()
  })

  it('the wait’s reasoning is only on the attempt the page is deciding on', async () => {
    serve({ n: 3, refused: 2, waiting: true })
    expect(await attemptTab(2)).toContain(WAIT_RAW)
    expect(await attemptTab(3)).not.toContain(WAIT_RAW)
    expect(await attemptTab(1)).not.toContain(WAIT_RAW)
  })

  it('Fix (6): the Run log section is RunLog’s own <details>, not wrapped in a second one', async () => {
    serve({ n: 2 })
    const log = section(parse(await attemptTab(1)), 'Run log')
    expect(log).toBeDefined()

  })

  it('Edge "Changes": each path relative to the repository, never under the worktree', async () => {

    expect(changes).toBeDefined()

    expect(said).not.toContain('.lingtai/worktrees')
    expect(said).not.toContain(runId(1))
  })

  it('Edge "Prompt": captioned by its size, its version in the title attribute only', async () => {
    serve({ n: 2 })
    const prompt = section(parse(await attemptTab(1)), 'Prompt')
    expect(prompt).toBeDefined()
    const summary = all(prompt!, (e) => e.tag === 'summary')[0]!
    expect(squash(text(summary))).toMatch(/4[.,]?\d*\s*(bytes|KB|kB|B)\b/)
    expect(text(summary)).not.toContain('ticket@191')
    expect(all(summary, (e) => (e.attrs.title ?? '').includes('ticket@191')).length + (summary.attrs.title?.includes('ticket@191') ? 1 : 0)).toBeGreaterThan(0)
  })

  it('Edge "The Events section": this attempt’s stream only, with no stream-id or seq heading', async () => {
    serve({ n: 2 })
    const events = section(parse(await attemptTab(1)), 'Events')
    expect(events).toBeDefined()

    expect(said).not.toContain('PROMPT-OF-ATTEMPT-2')
