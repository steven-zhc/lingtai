type Policy = {
  allowedDomains: string[]
  recipeDomains?: string[]
  allowRead: string[]
  allowWrite: string[]
  denyRead: string[]
  denyWrite: string[]
  weakerNetworkIsolation: boolean
  srtVersion: string
}

/** The set every policy of the demo shares. No path here is under `homedir()` or `stateDir()`. */
const common = (over: Partial<Policy> = {}): Policy => ({
  allowedDomains: BUILT_IN,
  recipeDomains: [],
  allowRead: ['/', '/etc/secrets/public'],
  allowWrite: ['/srv/wt', '/tmp', '/private/tmp'],
  denyRead: ['/etc/secrets'],
  denyWrite: ['/srv/wt/.git/hooks'],
  weakerNetworkIsolation: false,
  srtVersion: '0.0.83',

})

const keys = (commandId: string, step: string, action: string, runtime?: string) => ({
  step: step as 'implement',
  action,
  ...(runtime === undefined ? {} : { runtime }),
  runId: RUN,
  commandId,
})

type Boundary =
  | { kind: 'sandboxed'; srtVersion: string; policy: Policy }
  | { kind: 'unsandboxed'; reason: 'off by choice' }
  | { kind: 'unsandboxed'; reason: 'degraded'; why: string }
  | { kind: 'refused'; why: string }

const sandboxed = (policy: Policy): Boundary => ({ kind: 'sandboxed', srtVersion: policy.srtVersion, policy })

const entry = (host: string, count: number, firstAt: string, telemetry = false) => ({
  host,
  port: 443,
  count,
  firstAt,
  telemetry,
})

interface Inv {
  id: string
  step: string
  action: string
  runtime?: string
  boundary: Boundary
  /** undefined: no refusal record (pending). */
  refusals?:
    | { kind: 'entries'; entries: ReturnType<typeof entry>[]; complete: boolean }
    | { kind: 'unreadable'; reason: string }
}

function fold(invs: readonly Inv[], extra: (e: ReturnType<typeof makeStream>) => Envelope[] = () => []) {
  const e = makeStream(RUN)
  return reduceAttempt([
    e('RunStarted', {
      ticketId: `ticket-${PROJECT}-1`,
      runtime: 'claude-code',
      model: 'm',
      promptVersion: 'v',
      baseSha: 'b',
      configHash: 'c',
      worktree: '/srv/wt',
      invocation: null,
    }),
    ...invs.flatMap((i) => {
      const k = keys(i.id, i.step, i.action, i.runtime)
      const out: Envelope[] = [e('SandboxBoundaryRecorded', { ...k, boundary: i.boundary as never })]
      if (i.refusals !== undefined) out.push(e('SandboxConnectionsRefused', { ...k, read: i.refusals as never }))
      return out
    }),
    ...extra(e),
  ])
}

const none = { kind: 'entries' as const, entries: [], complete: true }

const viewOf = (invs: readonly Inv[], superseded = false) => sandboxViewOf(fold(invs), PROJECT, superseded)

const decode = (s: string) =>
  s
    .replace(/&#x27;|&#39;/g, "'")
    .replace(/&quot;/g, '"')
    .replace(/&lt;/g, '<')
    .replace(/&gt;/g, '>')
    .replace(/&nbsp;/g, ' ')
    .replace(/&amp;/g, '&')

/** The visible text: tags become spaces, entities are decoded, whitespace is collapsed. */
const textOf = (html: string) =>
  decode(html.replace(/<[^>]*>/g, ' '))
    .replace(/\s+/g, ' ')
    .trim()

/** The outer HTML of the element carrying `aria-label="<label>"`, or null. */
function part(html: string, label: string): string | null {
  const at = html.indexOf(`aria-label="${label}"`)
  if (at === -1) return null
  const start = html.lastIndexOf('<', at)
  const name = /^<([a-zA-Z0-9]+)/.exec(html.slice(start))![1]!
  const tag = new RegExp(`<(/?)${name}(?=[\\s>/])[^>]*>`, 'g')
  tag.lastIndex = start
  let depth = 0
  for (let m = tag.exec(html); m !== null; m = tag.exec(html)) {
    if (m[0].endsWith('/>')) continue
    depth += m[1] === '/' ? -1 : 1
    if (depth === 0) return html.slice(start, m.index + m[0].length)
  }
  return html.slice(start)
}

function mustPart(html: string, label: string): string {
  const p = part(html, label)
  expect(p, `a part labelled "${label}"`).not.toBeNull()
  return p!
}

/** The text of each `<tr>` inside `html`. */
const rowsOf = (html: string) =>
  [...html.matchAll(/<tr[\s>][\s\S]*?<\/tr>/g)].map((m) => textOf(m[0])).filter((t) => t !== '')

const count = (hay: string, needle: string) => hay.split(needle).length - 1

/** The text after `label` up to the next of `stops`, within `text`. */
function segment(text: string, label: string, stops: readonly string[]): string {
  const at = text.indexOf(label)
  expect(at, `"${label}" in the text`).toBeGreaterThan(-1)
  const rest = text.slice(at + label.length)
  const ends = stops.map((s) => rest.indexOf(` ${s} `)).filter((n) => n !== -1)
  return rest.slice(0, ends.length === 0 ? undefined : Math.min(...ends))
}

      refusals: none,

  })

    expect(segment(text, ' GitHub ', GROUPS)).toContain('*.githubusercontent.com')
    expect(segment(text, ' This recipe ', GROUPS)).toContain('none added')
  })

  it("lists the recipe's own hosts under This recipe, and a built-in host with no vendor under Other", () => {
    const p = common({ allowedDomains: [...BUILT_IN, 'example.org', 'mirror.example.net'], recipeDomains: ['example.org'] })
    const text = textOf(mustPart(tab(viewOf(demo().map((i) => ({ ...i, boundary: sandboxed(p) })))), REACH))
    expect(segment(text, ' This recipe ', GROUPS)).toContain('example.org')
    expect(segment(text, ' This recipe ', GROUPS)).not.toContain('none added')
    expect(segment(text, ' Other ', GROUPS)).toContain('mirror.example.net')
  })

  it('Edges, source not recorded: the host is listed under its vendor group, not under This recipe', () => {
    const { recipeDomains: _gone, ...older } = common({ allowedDomains: ['github.com', 'example.org'] })
    const text = textOf(mustPart(tab(viewOf(demo().map((i) => ({ ...i, boundary: sandboxed(older as Policy) })))), REACH))
    expect(segment(text, ' GitHub ', GROUPS)).toContain('github.com')
    expect(segment(text, ' This recipe ', GROUPS)).not.toContain('example.org')
    expect(segment(text, ' This recipe ', GROUPS)).not.toContain('github.com')
  })
})

describe('Want: short path names, the full path one hover away (Done when 5)', () => {
  const home = homedir()
  const lingtai = stateDir()
  const mirror = `${lingtai}/repos/${PROJECT}.git`
  const cache = `${lingtai}/cache/${PROJECT}`
  const named = () =>
    demo().map((i) => ({
      ...i,
      boundary: sandboxed(
        common({
          allowRead: ['/', cache],
          allowWrite: ['/srv/wt', mirror, cache, '/tmp', '/private/tmp', tmpdir()],
          denyRead: [`${home}/.ssh`, lingtai],
          denyWrite: [`${mirror}/hooks`],
        }),
      ),
    }))

  it("draws each path by its short name, longest prefix first, with the full path in the chip's title", () => {
    const reach = mustPart(tab(viewOf(named())), REACH)
    const chip = (short: string, full: string) =>
      new RegExp(`<[^>]*title="${full.replace(/[.*+?^${}()|[\]\\]/g, '\\$&')}"[^>]*>\\s*${short.replace(/[.*+?^${}()|[\]\\]/g, '\\$&')}\\s*<`)
    expect(reach).toMatch(chip('~/.ssh', `${home}/.ssh`))
    expect(reach).toMatch(chip('&lt;lingtai home&gt;', lingtai))
    expect(reach).toMatch(chip('&lt;mirror&gt;', mirror))
    expect(reach).toMatch(chip('&lt;cache&gt;', cache))
    // Longest prefix wins: a mirror path, not `<lingtai home>/repos/…`.
    expect(reach).toMatch(chip('&lt;mirror&gt;/hooks', `${mirror}/hooks`))
    expect(textOf(reach)).not.toContain('<lingtai home>/repos')
  })

  it('the /private twins are one chip, the shorter spelling, with both spellings in its title', () => {
    const reach = mustPart(tab(viewOf(demo())), REACH)
    expect(count(reach, '>/tmp<')).toBe(1)
    expect(count(reach, '>/private/tmp<')).toBe(0)
    const title = /title="([^"]*\/private\/tmp[^"]*)"/.exec(reach)?.[1]
    expect(title).toBeDefined()
    expect(title!.replace('/private/tmp', '')).toContain('/tmp')
  })

  it("Watch out: no $TMPDIR token, the board's temp directory stays a path, in a chip titled with it", () => {
    const html = tab(viewOf(named()))
    expect(html).not.toMatch(/TMPDIR|&lt;tmp|&lt;temp/i)
    const tmp = tmpdir().replace(/[.*+?^${}()|[\]\\]/g, '\\$&')
    expect(mustPart(html, REACH)).toMatch(new RegExp(`title="[^"]*${tmp}[^"]*"`))
  })

  it('the legend lists every short name used and its full path, and no name that is not used', () => {
    const legend = textOf(mustPart(tab(viewOf(named())), LEGEND))
    for (const [short, full] of [
      ['~', home],
      ['<lingtai home>', lingtai],
      ['<mirror>', mirror],
      ['<cache>', cache],
    ] as const) {
      expect(legend, short).toContain(short)
      expect(legend, short).toContain(full)
    }
    expect(legend).not.toContain('<worktree>')
  })
})

describe('Done when 5 (pure): the short name of a path', () => {
  type Places = { home: string; lingtaiHome: string; project: string; runId: string }
  const shortNameOf = (sandboxView as unknown as { shortNameOf?: (path: string, places: Places) => string })
    .shortNameOf
  const places: Places = {
    home: '/Users/x',
    lingtaiHome: '/Users/x/demo-lingtai',
    project: 'chapter-garden',
    runId: 'pass-c45',
  }

  it('is exported from apps/board/src/lib/sandbox-view.ts', () => {
    expect(typeof shortNameOf).toBe('function')
  })

  const cases: [string, string][] = [
    ['/Users/x', '~'],
    ['/Users/x/.ssh', '~/.ssh'],
    ['/Users/xy/file', '/Users/xy/file'],
    ['/Users/x/demo-lingtai', '<lingtai home>'],
    ['/Users/x/demo-lingtai/config.yml', '<lingtai home>/config.yml'],
    ['/Users/x/demo-lingtai/worktrees/chapter-garden/pass-c45', '<worktree>'],
    ['/Users/x/demo-lingtai/worktrees/chapter-garden/pass-c45/.git/hooks', '<worktree>/.git/hooks'],
    ['/Users/x/demo-lingtai/repos/chapter-garden.git', '<mirror>'],
    ['/Users/x/demo-lingtai/repos/chapter-garden.git/hooks', '<mirror>/hooks'],
    ['/Users/x/demo-lingtai/repos/chapter-garden.gitx', '<lingtai home>/repos/chapter-garden.gitx'],
    ['/Users/x/demo-lingtai/repos/other.git', '<lingtai home>/repos/other.git'],
    ['/Users/x/demo-lingtai/cache/chapter-garden', '<cache>'],
    ['/Users/x/demo-lingtai/cache/chapter-garden/pnpm', '<cache>/pnpm'],
    ['/opt/elsewhere', '/opt/elsewhere'],
    ['/tmp', '/tmp'],
    ['/var/folders/y7/b4/T', '/var/folders/y7/b4/T'],
  ]
  for (const [path, short] of cases) {
    it(`${path} → ${short}`, () => {
      expect(shortNameOf?.(path, places)).toBe(short)
    })
  }

  it('a Lingtai home outside the home directory is still <lingtai home>, never ~/…', () => {
    const away = { ...places, lingtaiHome: '/srv/lingtai' }
    expect(shortNameOf?.('/srv/lingtai/repos/chapter-garden.git/info', away)).toBe('<mirror>/info')
    expect(shortNameOf?.('/srv/lingtai', away)).toBe('<lingtai home>')
  })
})
